# Bushy thatch grass — accepted default (2026-07-03).  Uses ThatchGrass
# built-in default species; carpet shares the same geometry.
from dharmatiles.spec import Tile, Region, FloodFill, SurfaceConfig, D
from dharmatiles.layers import GrassCarpet, SoilCarpet
from dharmatiles.scatter import Grouped, Rocks, Uniform
from dharmatiles.grass.thatch import ThatchGrass, _default_species

_species = _default_species()

_mounds = SoilCarpet(
    n_blobs=30,
    blob_sigma=D.triangular(1.4, 1.9, 2.5),
    blob_h_scale=D[0.6:1.05],
    blob_power=2.5,
    blob_aspect=D[0.55:0.95],
    blob_shape_noise_amp=0.12,
    blob_jitter=0.85,
    blob_cluster_count=0,
    n_small=6,
    small_sigma=D[3.0:4.5],
    small_h=D[0.7:1.4],
    surface_texture_amp=0.0,
    surface_texture2_amp=0.0,
    edge_fade_mm=2.0,
)

tile = Tile(
    surface=SurfaceConfig(seed=1),
    areas=[
        Region(
            id='meadow',
            selector=FloodFill(0.5, 0.5),
            layers=[
                _mounds,   # first -> terrain_material = SOIL (brown ground)
                # Rocks BEFORE the carpet: carpet tube meshes are placed
                # around existing obstacles, not impaled by later rocks.
                # Crown rule: exposed height r*flat - sink must beat the grass
                # canopy near rocks: skirt 0.8 + stack 1.2 + blade 0.8 +
                # clearances ~0.6 = ~3.4 mm, plus margin.  min: 3.6*1.2-0.3 = 4.02.
                Rocks(placement=Uniform(count_per_square=3, gap_mm=7),
                      r=D[3.6:4.8].power(1.3),
                      flat=D[1.55:1.75],
                      n_cuts=5,
                      cut=D[0.82:0.96],
                      roughness=0.06,
                      sink=0.3),
                GrassCarpet(species=_species, placement=Grouped(groups_per_square=3)),
                ThatchGrass(),
            ],
        ),
    ],
)
